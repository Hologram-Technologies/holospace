\begin{lexlean}{HolospaceIsolation}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}

\importmodule{Foundation.Bytes}
\importmodule{Foundation.View.Text.V1.Model}
\title{Text application model}
\begin{semanticmodule}
definition{name: kindPrivate, parameters: [], result: Nat, body: 0}

definition{name: kindContent, parameters: [], result: Nat, body: 1}

definition{name: opRead, parameters: [], result: Nat, body: 0}

definition{name: opWrite, parameters: [], result: Nat, body: 1}

definition{name: answerAllow, parameters: [], result: Nat, body: 0}

definition{name: answerDeny, parameters: [], result: Nat, body: 1}

structure{
  name: Request,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: requester, type: Nat},
    {name: owner, type: Nat},
    {name: kind, type: Nat},
    {name: op, type: Nat}
  ]
}

definition{
  name: byteAt,
  parameters: [s : Bytes, i : Nat],
  result: (Option Nat),
  body: (match (prim index s i : (Option UInt8))
    | Option.none => (ctor Option.none [Nat])
    | Option.some b => (ctor Option.some [Nat] (prim to_nat b : Nat))),
  axioms: [Quot.sound, propext]
}

definition{
  name: digitsValue,
  parameters: [s : Bytes, i : Nat, acc : Nat, fuel : Nat],
  result: (Option Nat),
  body: (match fuel
    | Nat.zero => (ctor Option.some [Nat] acc)
    | Nat.succ rest => (match (byteAt s i)
      | Option.none => (ctor Option.none [Nat])
      | Option.some d => (if ((d <? 48) || (57 <? d)) then (ctor Option.none [Nat]) else (digitsValue s (i + 1) ((prim multiply acc 10 : Nat) + (prim subtract d 48 : Nat)) rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: parseDigits,
  parameters: [s : Bytes],
  result: (Option Nat),
  body: (if ((Foundation.Bytes::byteLength s) =? 0) then (ctor Option.none [Nat]) else (if (18 <? (Foundation.Bytes::byteLength s)) then (ctor Option.none [Nat]) else (match (byteAt s 0)
    | Option.none => (ctor Option.none [Nat])
    | Option.some lead => (if ((lead =? 48) && (1 <? (Foundation.Bytes::byteLength s))) then (ctor Option.none [Nat]) else (digitsValue s 0 0 (Foundation.Bytes::byteLength s)))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: parseNumber,
  parameters: [value : Bytes],
  result: (Option Nat),
  body: (parseDigits value),
  axioms: [Quot.sound, propext]
}

definition{
  name: parseRequester,
  parameters: [value : Bytes],
  result: (Option Nat),
  body: (match (parseNumber value)
    | Option.none => (ctor Option.none [Nat])
    | Option.some parsed => (if (1000000 <? parsed) then (ctor Option.none [Nat]) else (ctor Option.some [Nat] parsed))),
  axioms: [Quot.sound, propext]
}

definition{
  name: parseOwner,
  parameters: [value : Bytes],
  result: (Option Nat),
  body: (match (parseNumber value)
    | Option.none => (ctor Option.none [Nat])
    | Option.some parsed => (if (1000000 <? parsed) then (ctor Option.none [Nat]) else (ctor Option.some [Nat] parsed))),
  axioms: [Quot.sound, propext]
}

definition{
  name: parseKind,
  parameters: [value : Bytes],
  result: (Option Nat),
  body: (if (prim equal value 0x70726976617465 : Bool) then (ctor Option.some [Nat] (kindPrivate)) else (if (prim equal value 0x636f6e74656e74 : Bool) then (ctor Option.some [Nat] (kindContent)) else (ctor Option.none [Nat])))
}

definition{
  name: parseOp,
  parameters: [value : Bytes],
  result: (Option Nat),
  body: (if (prim equal value 0x72656164 : Bool) then (ctor Option.some [Nat] (opRead)) else (if (prim equal value 0x7772697465 : Bool) then (ctor Option.some [Nat] (opWrite)) else (ctor Option.none [Nat])))
}

definition{
  name: sliceOf,
  parameters: [s : Bytes, start : Nat, count : Nat],
  result: Bytes,
  body: (match (prim slice s start count : (Option Bytes))
    | Option.none => 0x
    | Option.some part => part)
}

definition{
  name: fieldEnd,
  parameters: [s : Bytes, i : Nat, fuel : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => i
    | Nat.succ rest => (match (byteAt s i)
      | Option.none => i
      | Option.some d => (if (d =? 32) then i else (fieldEnd s (i + 1) rest)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: decodeField4,
  parameters: [s : Bytes, start : Nat, v1 : Nat, v2 : Nat, v3 : Nat],
  result: (Option Request),
  body: (let e : Nat := (fieldEnd s start (Foundation.Bytes::byteLength s)); (match (parseOp (sliceOf s start (prim subtract e start : Nat)))
    | Option.none => (ctor Option.none [Request])
    | Option.some v4 => (if (e =? (Foundation.Bytes::byteLength s)) then (ctor Option.some [Request] (record Request {requester := v1, owner := v2, kind := v3, op := v4})) else (ctor Option.none [Request])))),
  axioms: [Quot.sound, propext]
}

definition{
  name: decodeField3,
  parameters: [s : Bytes, start : Nat, v1 : Nat, v2 : Nat],
  result: (Option Request),
  body: (let e : Nat := (fieldEnd s start (Foundation.Bytes::byteLength s)); (match (parseKind (sliceOf s start (prim subtract e start : Nat)))
    | Option.none => (ctor Option.none [Request])
    | Option.some v3 => (if (e <? (Foundation.Bytes::byteLength s)) then (decodeField4 s (e + 1) v1 v2 v3) else (ctor Option.none [Request])))),
  axioms: [Quot.sound, propext]
}

definition{
  name: decodeField2,
  parameters: [s : Bytes, start : Nat, v1 : Nat],
  result: (Option Request),
  body: (let e : Nat := (fieldEnd s start (Foundation.Bytes::byteLength s)); (match (parseOwner (sliceOf s start (prim subtract e start : Nat)))
    | Option.none => (ctor Option.none [Request])
    | Option.some v2 => (if (e <? (Foundation.Bytes::byteLength s)) then (decodeField3 s (e + 1) v1 v2) else (ctor Option.none [Request])))),
  axioms: [Quot.sound, propext]
}

definition{
  name: decodeField1,
  parameters: [s : Bytes, start : Nat],
  result: (Option Request),
  body: (let e : Nat := (fieldEnd s start (Foundation.Bytes::byteLength s)); (match (parseRequester (sliceOf s start (prim subtract e start : Nat)))
    | Option.none => (ctor Option.none [Request])
    | Option.some v1 => (if (e <? (Foundation.Bytes::byteLength s)) then (decodeField2 s (e + 1) v1) else (ctor Option.none [Request])))),
  axioms: [Quot.sound, propext]
}

definition{
  name: rule1,
  parameters: [request : Request],
  result: Bool,
  body: (((project request kind) =? (kindPrivate)) && ((project request owner) =? (project request requester)))
}

definition{
  name: rule2,
  parameters: [request : Request],
  result: Bool,
  body: (((project request kind) =? (kindContent)) && ((project request op) =? (opRead)))
}

definition{
  name: judge,
  parameters: [request : Request],
  result: Nat,
  body: (if (rule1 request) then (answerAllow) else (if (rule2 request) then (answerAllow) else (answerDeny)))
}

definition{
  name: encodeAnswer,
  parameters: [code : Nat],
  result: Bytes,
  body: (if (code =? (answerAllow)) then 0x616c6c6f77 else (if (code =? (answerDeny)) then 0x64656e79 else 0x696e76616c6964))
}

definition{
  name: dispatch,
  parameters: [value : Bytes],
  result: Bytes,
  body: (if (112 <? (Foundation.Bytes::byteLength value)) then 0x696e76616c6964 else (match (decodeField1 value 0)
    | Option.none => 0x696e76616c6964
    | Option.some request => (encodeAnswer (judge request)))),
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
